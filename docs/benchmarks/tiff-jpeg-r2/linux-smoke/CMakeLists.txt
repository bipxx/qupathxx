cmake_minimum_required(VERSION 3.24)
project(qpx_jpeg_codec_smoke C CXX)
list(APPEND CMAKE_MODULE_PATH "${QPX_SOURCE}/cmake")
include(CTest)
add_library(qupathxx_core STATIC empty.c)
include(TiffReader)
add_executable(codec probe.c)
target_link_libraries(codec PRIVATE qupathxx_core TIFF::tiff)
file(READ "${qpx_tiff_BINARY_DIR}/libtiff/tiffconf.h" tiff_config)
if(NOT tiff_config MATCHES "#define JPEG_SUPPORT 1" OR NOT HAVE_JPEGTURBO_DUAL_MODE_8_12)
  message(FATAL_ERROR "Pinned JPEG configuration was not admitted")
endif()
