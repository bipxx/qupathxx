cmake_minimum_required(VERSION 3.24)
project(qupathxx_annotation_geometry_host LANGUAGES CXX)
# Source-scoped local overlay over the unchanged FP04 SDK. The repository root
# owns normal integrated builds; this recipe keeps the frozen SDK recoverable.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC_VERSION STREQUAL "1944" OR
   NOT CMAKE_CXX_COMPILER_VERSION STREQUAL "19.44.35225.0")
  message(FATAL_ERROR "Use the recorded Windows x64 MSVC1944/compiler19.44.35225.0 SDK profile")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Native geometry configuration" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "The native geometry host requires Release")
endif()
get_target_property(base_core qupathxx::core IMPORTED_LOCATION_RELEASE)
file(SHA256 "${base_core}" base_core_sha)
if(NOT base_core_sha STREQUAL "4bf957c9ea6e46294c11b9bfb7d9c2496cf4fca607a9c28ba837768d301eb6bb")
  message(FATAL_ERROR "This overlay requires the exact retained FP04 SDK core; use root CMake for another native context")
endif()
get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(QUPATHXX_GEOMETRY_JSON_INCLUDE "${source_root}/third_party/native-overlay/json/include" CACHE PATH "Matching nlohmann-json3.12.0 headers")
set(QUPATHXX_GEOMETRY_SQLITE_INCLUDE "${source_root}/third_party/native-overlay/sqlite" CACHE PATH "Matching SQLite3.54.1 header")
foreach(header "${QUPATHXX_GEOMETRY_JSON_INCLUDE}/nlohmann/json.hpp" "${QUPATHXX_GEOMETRY_SQLITE_INCLUDE}/sqlite3.h")
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Provide the recorded native dependency header: ${header}")
  endif()
endforeach()
add_library(qupathxx_annotation_geometry_native STATIC
  "${source_root}/cpp/format_plugin_loader.cpp"
  "${source_root}/cpp/format_plugin_runtime.cpp"
  "${source_root}/cpp/format_import.cpp"
  "${source_root}/cpp/annotation_geometry_value.cpp")
set(QUPATHXX_GEOMETRY_MASK_OBJECT_DIR "" CACHE PATH "Optional exact retained native label/contour objects")
if(QUPATHXX_GEOMETRY_MASK_OBJECT_DIR)
  set(QPX_MASK_label_raster_SHA256 "0e3247feee2cc44361e01267bf58974aedc89c7968e17b9b34eb009264ccdfaa")
  set(QPX_MASK_tissue_contours_SHA256 "5e4ba627aa058cae385510856873ee796100aadfe3b94495a29f5d3bf3073559")
  foreach(name label_raster tissue_contours)
    set(object "${QUPATHXX_GEOMETRY_MASK_OBJECT_DIR}/${name}.obj")
    file(SHA256 "${object}" actual_sha)
    if(NOT actual_sha STREQUAL "${QPX_MASK_${name}_SHA256}")
      message(FATAL_ERROR "Native mask object identity differs: ${name}")
    endif()
    set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT YES GENERATED YES)
    target_sources(qupathxx_annotation_geometry_native PRIVATE "${object}")
  endforeach()
else()
  # Source-only rebuild of these existing authorities is also possible without
  # rebuilding the base SDK. Supply its matching GEOS3.15.0 public C header.
  set(QUPATHXX_GEOMETRY_GEOS_INCLUDE "" CACHE PATH "Matching GEOS3.15.0 C header directory")
  if(NOT EXISTS "${QUPATHXX_GEOMETRY_GEOS_INCLUDE}/geos_c.h")
    message(FATAL_ERROR "Provide matching GEOS header or exact retained native mask objects")
  endif()
  target_sources(qupathxx_annotation_geometry_native PRIVATE
    "${source_root}/cpp/label_raster.cpp" "${source_root}/cpp/tissue_contours.cpp")
  target_include_directories(qupathxx_annotation_geometry_native PRIVATE "${QUPATHXX_GEOMETRY_GEOS_INCLUDE}")
  target_compile_definitions(qupathxx_annotation_geometry_native PRIVATE QUPATHXX_HAS_GEOS)
endif()
target_include_directories(qupathxx_annotation_geometry_native PUBLIC "${source_root}/cpp/include"
  PRIVATE "${source_root}/cpp" "${QUPATHXX_GEOMETRY_JSON_INCLUDE}" "${QUPATHXX_GEOMETRY_SQLITE_INCLUDE}")
target_compile_definitions(qupathxx_annotation_geometry_native PRIVATE QUPATHXX_HAS_FORMAT_PLUGINS QUPATHXX_HAS_PROJECT)
target_compile_options(qupathxx_annotation_geometry_native PRIVATE /utf-8 /W4 /permissive-)
target_link_libraries(qupathxx_annotation_geometry_native PUBLIC qupathxx::core)
add_executable(qupathxx_format_import "${source_root}/cpp/format_import_main.cpp")
target_include_directories(qupathxx_format_import PRIVATE "${source_root}/cpp" "${QUPATHXX_GEOMETRY_JSON_INCLUDE}")
target_compile_options(qupathxx_format_import PRIVATE /utf-8 /W4 /permissive-)
target_link_libraries(qupathxx_format_import PRIVATE qupathxx_annotation_geometry_native)
qupathxx_stage_runtime(qupathxx_format_import)
