cmake_minimum_required(VERSION 3.24)
project(ngff_label_sdk_consumer LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(qpx_ngff_labels CONFIG REQUIRED)
add_executable(ngff_label_consumer main.cpp)
target_compile_features(ngff_label_consumer PRIVATE cxx_std_17)
target_compile_options(ngff_label_consumer PRIVATE /utf-8 /W4 /permissive-)
set_target_properties(ngff_label_consumer PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(ngff_label_consumer PRIVATE qupathxx::ngff_labels)
qupathxx_stage_runtime(ngff_label_consumer)
