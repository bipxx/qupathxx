cmake_minimum_required(VERSION 3.24)
project(qupathxx_annotation_plugin_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(qpx_annotation_plugin CONFIG REQUIRED)
add_executable(annotation_plugin_consumer main.cpp)
target_compile_options(annotation_plugin_consumer PRIVATE /utf-8 /permissive-)
target_link_libraries(annotation_plugin_consumer PRIVATE qpx_annotation_plugin::native)
qupathxx_stage_runtime(annotation_plugin_consumer)
