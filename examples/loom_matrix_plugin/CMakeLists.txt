cmake_minimum_required(VERSION 3.24)
project(loom_matrix_sdk_consumer LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(qpx_loom_matrix CONFIG REQUIRED)
add_executable(loom_matrix_consumer main.cpp)
target_compile_features(loom_matrix_consumer PRIVATE cxx_std_17)
target_compile_options(loom_matrix_consumer PRIVATE /utf-8 /W4 /permissive-)
set_target_properties(loom_matrix_consumer PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(loom_matrix_consumer PRIVATE qupathxx::loom_matrix)
qupathxx_stage_runtime(loom_matrix_consumer)
