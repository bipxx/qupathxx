cmake_minimum_required(VERSION 3.24)
project(qupathxx_sdk_consumer LANGUAGES CXX)
set(QUPATHXX_SDK_COMPONENTS core density_workflow nucls_import CACHE STRING "Required installed components")
find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS ${QUPATHXX_SDK_COMPONENTS})
# Repeated discovery must retain the same SDK/runtime configuration.
find_package(qupathxx 0.1.0 CONFIG REQUIRED)
add_executable(sdk_consumer main.cpp)
target_link_libraries(sdk_consumer PRIVATE qupathxx::core)
foreach(component density_workflow nucls_import)
  if(TARGET qupathxx::${component})
    target_link_libraries(sdk_consumer PRIVATE qupathxx::${component})
    target_compile_definitions(sdk_consumer PRIVATE QPX_SDK_TEST_${component})
  endif()
endforeach()
target_compile_options(sdk_consumer PRIVATE /utf-8 /showIncludes)
target_link_options(sdk_consumer PRIVATE /VERBOSE:LIB)
qupathxx_stage_runtime(sdk_consumer)

# Compile the entire installed public include surface, using its installed path.
get_target_property(sdk_includes qupathxx::core INTERFACE_INCLUDE_DIRECTORIES)
list(GET sdk_includes 0 sdk_include)
file(GLOB headers "${sdk_include}/qupathxx/*.hpp")
set(header_check "")
foreach(header IN LISTS headers)
  get_filename_component(name "${header}" NAME)
  string(APPEND header_check "#include <qupathxx/${name}>\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/installed_headers.cpp" "${header_check}")
add_library(sdk_headers OBJECT "${CMAKE_CURRENT_BINARY_DIR}/installed_headers.cpp")
target_link_libraries(sdk_headers PRIVATE qupathxx::core)
target_compile_options(sdk_headers PRIVATE /utf-8 /showIncludes)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/installed-targets-$<CONFIG>.txt" CONTENT
  "includes=$<TARGET_PROPERTY:qupathxx::core,INTERFACE_INCLUDE_DIRECTORIES>\ncore=$<TARGET_FILE:qupathxx::core>\nruntime=$<TARGET_PROPERTY:qupathxx::core,QUPATHXX_RUNTIME_FILES>\nsource_sha256=${qupathxx_SOURCE_SHA256}\n")

option(QUPATHXX_SDK_ABI_PROBES "Add excluded targets for expected ABI refusal checks" OFF)
if(QUPATHXX_SDK_ABI_PROBES)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/abi_probe.cpp" "int main(){return 0;}\n")
  add_executable(sdk_wrong_crt EXCLUDE_FROM_ALL "${CMAKE_CURRENT_BINARY_DIR}/abi_probe.cpp")
  target_link_libraries(sdk_wrong_crt PRIVATE qupathxx::core)
  set_property(TARGET sdk_wrong_crt PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
