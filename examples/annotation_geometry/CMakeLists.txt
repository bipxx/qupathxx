cmake_minimum_required(VERSION 3.24)
project(annotation_geometry_sdk_consumer LANGUAGES CXX)
find_package(qpx_annotation_geometry CONFIG REQUIRED)
add_executable(annotation_geometry_consumer main.cpp)
target_compile_features(annotation_geometry_consumer PRIVATE cxx_std_17)
target_compile_options(annotation_geometry_consumer PRIVATE /utf-8 /permissive-)
set_target_properties(annotation_geometry_consumer PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(annotation_geometry_consumer PRIVATE qpx_annotation_geometry::native)
qupathxx_stage_runtime(annotation_geometry_consumer)
