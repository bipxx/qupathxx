cmake_minimum_required(VERSION 3.24)
project(qupathxx VERSION 0.1.0 LANGUAGES CXX)
option(QUPATHXX_BUILD_PYTHON "Build nanobind bindings" ON)
option(QUPATHXX_BUILD_GUI "Build Dear ImGui desktop viewer" OFF)
option(QUPATHXX_BUILD_TESTS "Build native tests" OFF)
option(QUPATHXX_BUILD_OPENSLIDE "Enable OpenSlide image reader" OFF)
option(QUPATHXX_BUILD_GEOS "Enable robust polygon topology through GEOS" OFF)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
add_library(qupathxx_core STATIC cpp/core.cpp cpp/document_edits.cpp cpp/selection.cpp cpp/geometry.cpp cpp/spatial_index.cpp cpp/image_server.cpp cpp/openslide_server.cpp cpp/tile_scheduler.cpp cpp/view_tiles.cpp cpp/display.cpp)
find_package(Threads REQUIRED)
target_link_libraries(qupathxx_core PUBLIC Threads::Threads)
target_include_directories(qupathxx_core PUBLIC cpp/include)
target_compile_features(qupathxx_core PUBLIC cxx_std_17)
set_target_properties(qupathxx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(qupathxx_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(qupathxx_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(QUPATHXX_BUILD_GEOS)
  find_package(GEOS 3.12 CONFIG REQUIRED)
  target_link_libraries(qupathxx_core PRIVATE GEOS::geos_c)
  target_compile_definitions(qupathxx_core PRIVATE QUPATHXX_HAS_GEOS)
  find_file(GEOS_LICENSE_FILE NAMES COPYING HINTS "${GEOS_ROOT}/share/licenses/geos")
  if(WIN32 AND SKBUILD AND NOT EXISTS "${GEOS_LICENSE_FILE}")
    message(FATAL_ERROR "GEOS-enabled Windows wheels bundle its DLLs and require GEOS_LICENSE_FILE pointing to the GEOS COPYING file")
  endif()
  if(WIN32)
    install(FILES $<TARGET_FILE:GEOS::geos_c> $<TARGET_FILE:GEOS::geos> DESTINATION qupathxx)
  endif()
  if(EXISTS "${GEOS_LICENSE_FILE}")
    install(FILES "${GEOS_LICENSE_FILE}" DESTINATION qupathxx/licenses/geos RENAME COPYING)
  endif()
endif()
if(QUPATHXX_BUILD_OPENSLIDE)
  find_package(OpenSlide REQUIRED)
  target_link_libraries(qupathxx_core PRIVATE OpenSlide::OpenSlide)
  target_compile_definitions(qupathxx_core PRIVATE QUPATHXX_HAS_OPENSLIDE)
  if(WIN32)
    install(FILES "${OpenSlide_RUNTIME}" DESTINATION qupathxx)
    if(EXISTS "${OpenSlide_ROOT}/licenses")
      install(DIRECTORY "${OpenSlide_ROOT}/licenses/" DESTINATION qupathxx/licenses/openslide-sdk)
    endif()
  endif()
endif()
function(qupathxx_copy_runtime target)
  if(QUPATHXX_BUILD_GEOS AND WIN32)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      $<TARGET_FILE:GEOS::geos_c> $<TARGET_FILE:GEOS::geos> "$<TARGET_FILE_DIR:${target}>")
  endif()
  if(QUPATHXX_BUILD_OPENSLIDE AND WIN32)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${OpenSlide_RUNTIME}" "$<TARGET_FILE_DIR:${target}>")
  endif()
endfunction()
add_executable(qupathxx_region cpp/region_main.cpp)
target_link_libraries(qupathxx_region PRIVATE qupathxx_core)
qupathxx_copy_runtime(qupathxx_region)
install(TARGETS qupathxx_region RUNTIME DESTINATION qupathxx)
if(QUPATHXX_BUILD_GUI)
  include(FetchContent)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.9b GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(glfw imgui)
  find_package(OpenGL REQUIRED)
  add_library(qupathxx_gui STATIC cpp/gui.cpp
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(qupathxx_gui PRIVATE ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(qupathxx_gui PUBLIC qupathxx_core PRIVATE glfw OpenGL::GL)
  set_target_properties(qupathxx_gui PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_executable(qupathxx_app cpp/main.cpp)
  target_link_libraries(qupathxx_app PRIVATE qupathxx_gui)
  qupathxx_copy_runtime(qupathxx_app)
  install(TARGETS qupathxx_app RUNTIME DESTINATION qupathxx)
endif()
if(QUPATHXX_BUILD_PYTHON)
  find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
    OUTPUT_VARIABLE nanobind_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_package(nanobind CONFIG REQUIRED)
  nanobind_add_module(_native cpp/bindings.cpp cpp/scheduler_bindings.cpp)
  target_link_libraries(_native PRIVATE qupathxx_core)
  qupathxx_copy_runtime(_native)
  if(QUPATHXX_BUILD_GUI)
    target_link_libraries(_native PRIVATE qupathxx_gui)
    target_compile_definitions(_native PRIVATE QUPATHXX_HAS_GUI)
  endif()
  install(TARGETS _native LIBRARY DESTINATION qupathxx RUNTIME DESTINATION qupathxx)
endif()
