      message(FATAL_ERROR "Bundling OpenSlide requires the self-contained SDK and its licenses")
    endif()
    get_filename_component(QPX_OPENSLIDE_ELF "${OpenSlide_LIBRARY}" REALPATH)
    install(FILES "${QPX_OPENSLIDE_ELF}" DESTINATION qupathxx RENAME libopenslide.so.1)
    install(DIRECTORY "${OpenSlide_ROOT}/licenses/" DESTINATION qupathxx/licenses/openslide-sdk)
  endif()
endif()
function(qupathxx_copy_runtime target)
  if(QUPATHXX_BUILD_GEOS AND WIN32)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      $<TARGET_FILE:GEOS::geos_c> $<TARGET_FILE:GEOS::geos> "$<TARGET_FILE_DIR:${target}>")
  endif()
  if(QUPATHXX_BUNDLE_OPENSLIDE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_target_properties(${target} PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  if(QUPATHXX_BUILD_OPENSLIDE AND WIN32)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${OpenSlide_RUNTIME}" "$<TARGET_FILE_DIR:${target}>")
  endif()
endfunction()

# Record the actual checked-out source bytes, independently of Git cleanliness.
file(GLOB_RECURSE QPX_BUILD_INPUTS CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/cpp/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/cpp/*.hpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*.in"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.py")
list(APPEND QPX_BUILD_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/pyproject.toml")
list(SORT QPX_BUILD_INPUTS)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${QPX_BUILD_INPUTS})
set(QPX_SOURCE_MANIFEST "")
foreach(input IN LISTS QPX_BUILD_INPUTS)
  file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${input}")
  file(SHA256 "${input}" digest)
  string(APPEND QPX_SOURCE_MANIFEST "${relative}=${digest}\n")
endforeach()
string(SHA256 QPX_SOURCE_SHA256 "${QPX_SOURCE_MANIFEST}")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE QPX_GIT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT QPX_GIT_REVISION)
  set(QPX_GIT_REVISION "unknown")
endif()
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE QPX_GIT_STATUS ERROR_QUIET)
if(QPX_GIT_STATUS AND NOT QPX_GIT_REVISION STREQUAL "unknown")
  string(APPEND QPX_GIT_REVISION "+dirty")
endif()
set(QPX_ENV_GUI ${QUPATHXX_BUILD_GUI})
set(QPX_ENV_PYTHON ${QUPATHXX_BUILD_PYTHON})
set(QPX_ENV_INTERCHANGE ${QUPATHXX_BUILD_INTERCHANGE})
configure_file(cmake/build_manifest.hpp.in generated/build_manifest.hpp @ONLY)
target_include_directories(qupathxx_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_property(SOURCE cpp/provenance.cpp APPEND PROPERTY COMPILE_DEFINITIONS "QPX_BUILD_CONFIGURATION=\"$<CONFIG>\"")

if(MSVC)
  set_property(SOURCE cpp/provenance.cpp APPEND PROPERTY COMPILE_OPTIONS /utf-8)
endif()
