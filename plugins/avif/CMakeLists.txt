set(inputs "${CMAKE_CURRENT_BINARY_DIR}/codec-inputs")

function(unpack_decoder component expected)
  set(archive "${QUPATHXX_AVIF_ARCHIVES}/${component}-decoder-sources.tar.gz")
  if(NOT EXISTS "${archive}")
    message(FATAL_ERROR "Provide the pinned offline source selection for ${component}")
  endif()
  file(SHA256 "${archive}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Unexpected decoder source identity for ${component}")
  endif()
  if(NOT EXISTS "${inputs}/${expected}.extracted")
    file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${inputs}")
    file(WRITE "${inputs}/${expected}.extracted" "${actual}")
  endif()
endfunction()
