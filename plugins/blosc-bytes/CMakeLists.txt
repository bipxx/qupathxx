cmake_minimum_required(VERSION 3.24)
project(qupathxx_blosc_bytes_plugin VERSION 1.0.0 LANGUAGES C CXX)

set(QUPATHXX_READER_PLUGIN_KIT_PREFIX "" CACHE PATH "Relocated ReaderPluginKit prefix")
if(NOT IS_DIRECTORY "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}")
  message(FATAL_ERROR "Set QUPATHXX_READER_PLUGIN_KIT_PREFIX to the existing matching kit; no dependency downloads")
endif()
set(qpx_kit "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/share/qupathxx/reader-plugin")
include("${qpx_kit}/profile.cmake")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
   NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "This plugin requires Windows x64 MSVC")
endif()
if(NOT MSVC_VERSION STREQUAL QUPATHXX_READER_PLUGIN_MSVC_VERSION OR
   NOT CMAKE_CXX_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the C++ compiler recorded by the ReaderPluginKit")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Supported configuration" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "This plugin requires CMAKE_BUILD_TYPE=Release")
endif()
if(CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreadedDLL")
  message(FATAL_ERROR "This plugin requires the Release dynamic CRT (/MD)")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
configure_file(reader_plugin_profile.hpp.in reader_plugin_profile.hpp @ONLY)
set(QUPATHXX_BLOSC_SOURCE_DIR "" CACHE PATH "Verified C-Blosc 1.21.6 source archive, extracted")
if(NOT CMAKE_C_COMPILER_ID STREQUAL "MSVC" OR
   NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the C compiler recorded by the matching ReaderPluginKit")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" lock)
string(JSON count LENGTH "${lock}" members)
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  string(JSON member MEMBER "${lock}" members ${i})
  string(JSON expected GET "${lock}" members "${member}")
  if(NOT EXISTS "${QUPATHXX_BLOSC_SOURCE_DIR}/${member}")
    message(FATAL_ERROR "Missing pinned Blosc input: ${member}")
  endif()
  file(SHA256 "${QUPATHXX_BLOSC_SOURCE_DIR}/${member}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Changed pinned Blosc input: ${member}")
  endif()
endforeach()
set(blosc_sources)
foreach(member blosc.c blosclz.c fastcopy.c shuffle-generic.c bitshuffle-generic.c shuffle.c)
  list(APPEND blosc_sources "${QUPATHXX_BLOSC_SOURCE_DIR}/blosc/${member}")
endforeach()
add_library(qpx_blosc STATIC ${blosc_sources}
  "${QUPATHXX_BLOSC_SOURCE_DIR}/internal-complibs/lz4-1.9.4/lz4.c"
  "${QUPATHXX_BLOSC_SOURCE_DIR}/internal-complibs/lz4-1.9.4/lz4hc.c")
target_include_directories(qpx_blosc PUBLIC "${QUPATHXX_BLOSC_SOURCE_DIR}/blosc"
  PRIVATE "${QUPATHXX_BLOSC_SOURCE_DIR}/internal-complibs/lz4-1.9.4")
target_compile_definitions(qpx_blosc PRIVATE HAVE_LZ4 _CRT_SECURE_NO_WARNINGS)
add_library(qpx_blosc_bytes_plugin MODULE blosc_reader.cpp)
set_target_properties(qpx_blosc_bytes_plugin PROPERTIES PREFIX "")
target_compile_definitions(qpx_blosc_bytes_plugin PRIVATE QPX_READER_PLUGIN_EXPORTS QPX_READER_PLUGIN_SOURCE_FILES _CRT_SECURE_NO_WARNINGS)
target_compile_options(qpx_blosc_bytes_plugin PRIVATE /utf-8 /W4 /permissive-)
target_include_directories(qpx_blosc_bytes_plugin PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(qpx_blosc_bytes_plugin PRIVATE qpx_blosc)
add_custom_command(TARGET qpx_blosc_bytes_plugin POST_BUILD COMMAND "${CMAKE_COMMAND}"
  "-DMODULE_FILE=$<TARGET_FILE:qpx_blosc_bytes_plugin>"
  "-DPROFILE=windows-x64-msvc${QUPATHXX_READER_PLUGIN_MSVC_VERSION}-release-md"
  "-DOUTPUT=$<TARGET_FILE_DIR:qpx_blosc_bytes_plugin>/reader-plugins.json"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/write_manifest.cmake" VERBATIM)
install(TARGETS qpx_blosc_bytes_plugin RUNTIME DESTINATION . LIBRARY DESTINATION . CONFIGURATIONS Release)
install(FILES "$<TARGET_FILE_DIR:qpx_blosc_bytes_plugin>/reader-plugins.json" source-lock.json THIRD_PARTY_NOTICES.txt DESTINATION . CONFIGURATIONS Release)
install(FILES "${qpx_kit}/LICENSE" DESTINATION licenses/qupathxx)
