cmake_minimum_required(VERSION 3.24)
project(qupathxx_mrc_reader_plugin VERSION 1.5.0 LANGUAGES C CXX)

set(qpx_image_target qpx_mrc_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/mrc_reader.cpp")
set(qpx_image_id org.qupathxx.mrc)
set(qpx_image_name MRC2014)
set(qpx_image_contract mrc2014-scalar-window/3)
include("../image-common/plugin_build.cmake")

target_compile_definitions(qpx_mrc_plugin PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
  QPX_MRC_VERSION="${PROJECT_VERSION}" QPX_MRC_CONTRACT="${qpx_image_contract}")

set(QUPATHXX_MRC_ZLIB_ARCHIVE "" CACHE FILEPATH "Admitted offline zlib 1.3.2 source archive")
if(NOT EXISTS "${QUPATHXX_MRC_ZLIB_ARCHIVE}")
  message(FATAL_ERROR "Provide the admitted offline zlib 1.3.2 source archive.")
endif()
file(SHA256 "${QUPATHXX_MRC_ZLIB_ARCHIVE}" zlib_digest)
if(NOT zlib_digest STREQUAL "bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16")
  message(FATAL_ERROR "MRC requires the exact admitted zlib archive")
endif()
if(NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the C compiler recorded by the ReaderPluginKit")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(zlib_root "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs")
set(zlib_source "${zlib_root}/zlib-1.3.2")
if(NOT EXISTS "${zlib_source}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_MRC_ZLIB_ARCHIVE}" DESTINATION "${zlib_root}")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
add_subdirectory("${zlib_source}" "${CMAKE_CURRENT_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
set(zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/include")
configure_file("${zlib_source}/zlib.h" "${zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${zlib_headers}/zconf.h" COPYONLY)
file(READ "${zlib_headers}/zconf.h" zlib_config)
if(NOT zlib_config MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "MRC requires private prefixed zlib symbols")
endif()
target_include_directories(qpx_mrc_plugin PRIVATE "${zlib_headers}")
target_link_libraries(qpx_mrc_plugin PRIVATE zlibstatic)
install(FILES "${zlib_source}/LICENSE" DESTINATION licenses/zlib)

set(QUPATHXX_MRC_BZIP2_ARCHIVE "" CACHE FILEPATH "Admitted offline bzip2 1.0.8 source archive")
if(NOT EXISTS "${QUPATHXX_MRC_BZIP2_ARCHIVE}")
  message(FATAL_ERROR "Provide the admitted bzip2 1.0.8 source archive; no dependency downloads")
endif()
file(SHA256 "${QUPATHXX_MRC_BZIP2_ARCHIVE}" bzip2_digest)
if(NOT bzip2_digest STREQUAL "ab5a03176ee106d3f0fa90e381da478ddae405918153cca248e682cd0c4a2269")
  message(FATAL_ERROR "MRC requires the exact admitted bzip2 archive")
endif()
set(bzip2_dir "${CMAKE_CURRENT_BINARY_DIR}/bzip2-inputs")
set(bzip2_sources "${bzip2_dir}/bzip2-1.0.8")
if(NOT EXISTS "${bzip2_sources}/bzlib.h")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_MRC_BZIP2_ARCHIVE}" DESTINATION "${bzip2_dir}")
endif()
set(bzip2_files blocksort.c huffman.c crctable.c randtable.c compress.c decompress.c bzlib.c)
list(TRANSFORM bzip2_files PREPEND "${bzip2_sources}/")
add_library(qpx_mrc_bzip2 STATIC ${bzip2_files})
target_compile_definitions(qpx_mrc_bzip2 PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(qpx_mrc_bzip2 PUBLIC "${bzip2_sources}")
target_link_libraries(qpx_mrc_plugin PRIVATE qpx_mrc_bzip2)
install(FILES "${bzip2_sources}/LICENSE" DESTINATION licenses/bzip2)

option(QUPATHXX_MRC_BUILD_TESTS "Build independent C ABI and native SDK consumers" OFF)
if(QUPATHXX_MRC_BUILD_TESTS)
  add_executable(mrc_plugin_tests tests/mrc_plugin_tests.cpp)
  target_include_directories(mrc_plugin_tests PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include")
  target_compile_options(mrc_plugin_tests PRIVATE /utf-8 /W4 /permissive-)
  target_compile_definitions(mrc_plugin_tests PRIVATE
    QPX_MRC_VERSION="${PROJECT_VERSION}" QPX_MRC_CONTRACT="${qpx_image_contract}")

  find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
  add_executable(mrc_project_consumer tests/mrc_project_consumer.cpp)
  target_compile_options(mrc_project_consumer PRIVATE /utf-8 /W4 /permissive-)
  target_link_libraries(mrc_project_consumer PRIVATE qupathxx::core)
  qupathxx_stage_runtime(mrc_project_consumer)
endif()
