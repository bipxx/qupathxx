cmake_minimum_required(VERSION 3.24)
project(qupathxx_openexr_reader VERSION 1.1.0 LANGUAGES C CXX)
set(qpx_image_target qpx_openexr_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/openexr_reader.cpp")
set(qpx_image_id org.qupathxx.openexr)
set(qpx_image_name OpenEXR)
set(qpx_image_contract openexr-linear-rgb-scanline/1)
include("../image-common/plugin_build.cmake")
target_compile_definitions(qpx_openexr_plugin PRIVATE QPX_OPENEXR_VERSION="${PROJECT_VERSION}")

set(QUPATHXX_OPENEXR_ZLIB_ARCHIVE "" CACHE FILEPATH "Admitted offline zlib 1.3.2 source archive")
if(NOT EXISTS "${QUPATHXX_OPENEXR_ZLIB_ARCHIVE}")
  message(FATAL_ERROR "Supply the admitted zlib 1.3.2 archive")
endif()
file(SHA256 "${QUPATHXX_OPENEXR_ZLIB_ARCHIVE}" zlib_digest)
if(NOT zlib_digest STREQUAL "bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16")
  message(FATAL_ERROR "Unexpected zlib source archive")
endif()
set(zlib "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs/zlib-1.3.2")
if(NOT EXISTS "${zlib}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_OPENEXR_ZLIB_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
add_subdirectory("${zlib}" "${CMAKE_CURRENT_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
set(zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/include")
configure_file("${zlib}/zlib.h" "${zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${zlib_headers}/zconf.h" COPYONLY)
file(READ "${zlib_headers}/zconf.h" zlib_config)
if(NOT zlib_config MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "OpenEXR requires private prefixed zlib symbols")
endif()
target_include_directories(qpx_openexr_plugin PRIVATE "${zlib_headers}")
target_link_libraries(qpx_openexr_plugin PRIVATE zlibstatic)
install(FILES "${zlib}/LICENSE" DESTINATION licenses/zlib)
