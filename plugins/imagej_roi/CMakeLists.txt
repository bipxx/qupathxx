cmake_minimum_required(VERSION 3.24)
project(qupathxx_imagej_roi_plugin VERSION 1.1.0 LANGUAGES C CXX)
set(qpx_provider_target qpx_imagej_roi_plugin)
set(qpx_provider_source plugin.cpp)
set(qpx_provider_id org.qupathxx.imagej-roi)
set(qpx_format_id imagej-roi)
set(qpx_provider_domain annotation)
set(qpx_provider_abi 3)
set(qpx_provider_name "ImageJ ROI and ZIP")
set(qpx_provider_contract imagej-pixel-roi/1)
set(qpx_provider_version "${PROJECT_VERSION}")
include("${CMAKE_CURRENT_SOURCE_DIR}/../format-common/plugin_build.cmake")
target_sources(qpx_imagej_roi_plugin PRIVATE roi.cpp zip.cpp)
set(QUPATHXX_IMAGEJ_ZLIB_SOURCE_DIR "" CACHE PATH "Admitted zlib1.3.2 source directory (no downloads)")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/zlib-source-lock.json" zlib_lock)
string(JSON zlib_count LENGTH "${zlib_lock}" members)
math(EXPR zlib_last "${zlib_count}-1")
foreach(index RANGE 0 ${zlib_last})
 string(JSON member MEMBER "${zlib_lock}" members ${index})
 string(JSON expected GET "${zlib_lock}" members "${member}" sha256)
 file(SHA256 "${QUPATHXX_IMAGEJ_ZLIB_SOURCE_DIR}/${member}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Changed or missing admitted zlib source: ${member}")
 endif()
endforeach()
if(NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_FORMAT_PLUGIN_CXX_COMPILER_VERSION)
 message(FATAL_ERROR "Use the C compiler recorded by the FormatPluginKit")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
foreach(name GVMAT64 INFBACK9 CRC32VX)
 set(ZLIB_WITH_${name} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(name ADA BLAST IOSTREAM3 MINIZIP PUFF TESTZLIB ZLIB1_DLL)
 set(ZLIB_BUILD_${name} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${QUPATHXX_IMAGEJ_ZLIB_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/zlib")
set(imagej_zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/imagej-include")
file(MAKE_DIRECTORY "${imagej_zlib_headers}")
configure_file("${QUPATHXX_IMAGEJ_ZLIB_SOURCE_DIR}/zlib.h" "${imagej_zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${imagej_zlib_headers}/zconf.h" COPYONLY)
file(READ "${imagej_zlib_headers}/zconf.h" zconf)
if(NOT zconf MATCHES "#define Z_PREFIX 1")
 message(FATAL_ERROR "ImageJ ZIP requires private prefixed compression symbols")
endif()
target_include_directories(qpx_imagej_roi_plugin PRIVATE "${imagej_zlib_headers}")
target_link_libraries(qpx_imagej_roi_plugin PRIVATE zlibstatic)
install(FILES "${QUPATHXX_IMAGEJ_ZLIB_SOURCE_DIR}/LICENSE" DESTINATION licenses/zlib)
