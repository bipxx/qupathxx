cmake_minimum_required(VERSION 3.24)
project(qupathxx_nrrd_reader_plugin VERSION 1.14.0 LANGUAGES C CXX)
set(qpx_nrrd_contract nrrd-scalar-window/15)

set(QUPATHXX_READER_PLUGIN_KIT_PREFIX "" CACHE PATH "Relocated ReaderPluginKit prefix")
option(QUPATHXX_NRRD_BUILD_TESTS "Build independent ABI and installed-SDK checks" OFF)
if(NOT IS_DIRECTORY "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}")
  message(FATAL_ERROR "Set QUPATHXX_READER_PLUGIN_KIT_PREFIX to the existing matching kit; no dependency downloads")
endif()
set(qpx_kit "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/share/qupathxx/reader-plugin")
include("${qpx_kit}/profile.cmake")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
   NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "This plugin requires Windows x64 MSVC")
endif()
if(NOT MSVC_VERSION STREQUAL QUPATHXX_READER_PLUGIN_MSVC_VERSION OR
   NOT CMAKE_CXX_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION OR
   NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the MSVC compiler recorded by the ReaderPluginKit")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Supported configuration" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "This plugin requires CMAKE_BUILD_TYPE=Release")
endif()
if(CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreadedDLL")
  message(FATAL_ERROR "This plugin requires the Release dynamic CRT (/MD)")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)
configure_file(reader_plugin_profile.hpp.in reader_plugin_profile.hpp @ONLY)
set(QUPATHXX_NRRD_ZLIB_ARCHIVE "" CACHE FILEPATH "Admitted offline zlib 1.3.2 source archive")
if(NOT EXISTS "${QUPATHXX_NRRD_ZLIB_ARCHIVE}")
  message(FATAL_ERROR "Provide the admitted zlib 1.3.2 source archive; no dependency downloads")
endif()
file(SHA256 "${QUPATHXX_NRRD_ZLIB_ARCHIVE}" zlib_digest)
if(NOT zlib_digest STREQUAL "bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16")
  message(FATAL_ERROR "NRRD requires the exact admitted zlib archive")
endif()
set(zlib_sources "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs")
if(NOT EXISTS "${zlib_sources}/zlib-1.3.2/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_NRRD_ZLIB_ARCHIVE}" DESTINATION "${zlib_sources}")
endif()
set(zlib_sources "${zlib_sources}/zlib-1.3.2")
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
add_subdirectory("${zlib_sources}" "${CMAKE_CURRENT_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
set(zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/include")
configure_file("${zlib_sources}/zlib.h" "${zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${zlib_headers}/zconf.h" COPYONLY)
file(READ "${zlib_headers}/zconf.h" zlib_config)
if(NOT zlib_config MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "NRRD requires private prefixed zlib symbols")
endif()
set(QUPATHXX_NRRD_BZIP2_ARCHIVE "" CACHE FILEPATH "Admitted offline bzip2 1.0.8 source archive")
if(NOT EXISTS "${QUPATHXX_NRRD_BZIP2_ARCHIVE}")
  message(FATAL_ERROR "Provide the admitted bzip2 1.0.8 source archive; no dependency downloads")
endif()
file(SHA256 "${QUPATHXX_NRRD_BZIP2_ARCHIVE}" bzip2_digest)
if(NOT bzip2_digest STREQUAL "ab5a03176ee106d3f0fa90e381da478ddae405918153cca248e682cd0c4a2269")
  message(FATAL_ERROR "NRRD requires the exact admitted bzip2 archive")
endif()
set(bzip2_dir "${CMAKE_CURRENT_BINARY_DIR}/bzip2-inputs")
set(bzip2_sources "${bzip2_dir}/bzip2-1.0.8")
if(NOT EXISTS "${bzip2_sources}/bzlib.h")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_NRRD_BZIP2_ARCHIVE}" DESTINATION "${bzip2_dir}")
endif()
set(bzip2_files blocksort.c huffman.c crctable.c randtable.c compress.c decompress.c bzlib.c)
list(TRANSFORM bzip2_files PREPEND "${bzip2_sources}/")
add_library(qpx_nrrd_bzip2 STATIC ${bzip2_files})
target_compile_definitions(qpx_nrrd_bzip2 PRIVATE _CRT_SECURE_NO_WARNINGS)
target_include_directories(qpx_nrrd_bzip2 PUBLIC "${bzip2_sources}")
add_library(qpx_nrrd_plugin MODULE nrrd_reader.cpp)
set_target_properties(qpx_nrrd_plugin PROPERTIES PREFIX "")
target_compile_definitions(qpx_nrrd_plugin PRIVATE
  QPX_READER_PLUGIN_EXPORTS QPX_READER_PLUGIN_SOURCE_FILES _CRT_SECURE_NO_WARNINGS
  NOMINMAX WIN32_LEAN_AND_MEAN
  QPX_NRRD_VERSION="${PROJECT_VERSION}" QPX_NRRD_CONTRACT="${qpx_nrrd_contract}")
target_compile_options(qpx_nrrd_plugin PRIVATE /utf-8 /W4 /permissive-)
target_include_directories(qpx_nrrd_plugin PRIVATE
  "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include" "${CMAKE_CURRENT_BINARY_DIR}" "${zlib_headers}")
target_link_libraries(qpx_nrrd_plugin PRIVATE zlibstatic qpx_nrrd_bzip2)
add_custom_command(TARGET qpx_nrrd_plugin POST_BUILD COMMAND "${CMAKE_COMMAND}"
  "-DMODULE_FILE=$<TARGET_FILE:qpx_nrrd_plugin>"
  "-DVERSION=${PROJECT_VERSION}" "-DCONTRACT=${qpx_nrrd_contract}"
  "-DPROFILE=windows-x64-msvc${QUPATHXX_READER_PLUGIN_MSVC_VERSION}-release-md"
  "-DOUTPUT=$<TARGET_FILE_DIR:qpx_nrrd_plugin>/reader-plugins.json"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/write_manifest.cmake" VERBATIM)
install(TARGETS qpx_nrrd_plugin RUNTIME DESTINATION . LIBRARY DESTINATION . CONFIGURATIONS Release)
install(FILES "$<TARGET_FILE_DIR:qpx_nrrd_plugin>/reader-plugins.json" DESTINATION . CONFIGURATIONS Release)
install(FILES "${qpx_kit}/LICENSE" DESTINATION licenses/qupathxx)
install(FILES "${zlib_sources}/LICENSE" DESTINATION licenses/zlib)
install(FILES "${bzip2_sources}/LICENSE" DESTINATION licenses/bzip2)

if(QUPATHXX_NRRD_BUILD_TESTS)
  add_executable(nrrd_plugin_tests tests/nrrd_plugin_tests.cpp)
  target_include_directories(nrrd_plugin_tests PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include")
  target_compile_options(nrrd_plugin_tests PRIVATE /utf-8 /permissive-)
  target_compile_definitions(nrrd_plugin_tests PRIVATE
    QPX_NRRD_VERSION="${PROJECT_VERSION}" QPX_NRRD_CONTRACT="${qpx_nrrd_contract}")
  find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
  add_executable(nrrd_project_consumer tests/nrrd_project_consumer.cpp)
  target_compile_options(nrrd_project_consumer PRIVATE /utf-8 /permissive-)
  target_link_libraries(nrrd_project_consumer PRIVATE qupathxx::core)
  qupathxx_stage_runtime(nrrd_project_consumer)
endif()
