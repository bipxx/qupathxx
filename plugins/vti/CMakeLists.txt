cmake_minimum_required(VERSION 3.24)
project(qupathxx_vti_reader VERSION 1.8.0 LANGUAGES C CXX)

set(qpx_image_target qpx_vti_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/vti_reader.cpp")
set(qpx_image_id org.qupathxx.vti)
set(qpx_image_name "VTK ImageData")
set(qpx_image_contract vtk-imagedata-point-array/3)
include("../image-common/plugin_build.cmake")
set(QUPATHXX_VTI_TINYXML2_ARCHIVE "" CACHE FILEPATH "Admitted offline tinyxml2 11.0.0 source archive")
if(NOT EXISTS "${QUPATHXX_VTI_TINYXML2_ARCHIVE}")
  message(FATAL_ERROR "Supply the admitted tinyxml2 11.0.0 archive")
endif()
file(SHA256 "${QUPATHXX_VTI_TINYXML2_ARCHIVE}" source_digest)
if(NOT source_digest STREQUAL "5556deb5081fb246ee92afae73efd943c889cef0cafea92b0b82422d6a18f289")
  message(FATAL_ERROR "Unexpected tinyxml2 source archive")
endif()
set(tinyxml "${CMAKE_CURRENT_BINARY_DIR}/xml-inputs/tinyxml2-11.0.0")
if(NOT EXISTS "${tinyxml}/tinyxml2.cpp")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_VTI_TINYXML2_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/xml-inputs")
endif()
target_sources(qpx_vti_plugin PRIVATE "${tinyxml}/tinyxml2.cpp")
target_include_directories(qpx_vti_plugin PRIVATE "${tinyxml}")
target_compile_definitions(qpx_vti_plugin PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
  QPX_VTI_VERSION="${PROJECT_VERSION}" QPX_VTI_CONTRACT="${qpx_image_contract}")
install(FILES "${tinyxml}/LICENSE.txt" DESTINATION licenses/tinyxml2)

set(QUPATHXX_VTI_ZLIB_ARCHIVE "" CACHE FILEPATH "Admitted offline zlib 1.3.2 source archive")
if(NOT EXISTS "${QUPATHXX_VTI_ZLIB_ARCHIVE}")
  message(FATAL_ERROR "Supply the admitted zlib 1.3.2 archive")
endif()
file(SHA256 "${QUPATHXX_VTI_ZLIB_ARCHIVE}" zlib_digest)
if(NOT zlib_digest STREQUAL "bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16")
  message(FATAL_ERROR "Unexpected zlib source archive")
endif()
set(zlib "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs/zlib-1.3.2")
if(NOT EXISTS "${zlib}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_VTI_ZLIB_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
add_subdirectory("${zlib}" "${CMAKE_CURRENT_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
set(zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/include")
configure_file("${zlib}/zlib.h" "${zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${zlib_headers}/zconf.h" COPYONLY)
file(READ "${zlib_headers}/zconf.h" zlib_config)
if(NOT zlib_config MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "VTI requires private prefixed zlib symbols")
endif()
target_include_directories(qpx_vti_plugin PRIVATE "${zlib_headers}")
target_link_libraries(qpx_vti_plugin PRIVATE zlibstatic)
install(FILES "${zlib}/LICENSE" DESTINATION licenses/zlib)

set(QUPATHXX_VTI_LZ4_ARCHIVE "" CACHE FILEPATH "Admitted offline LZ4 1.10.0 library source ZIP")
if(NOT EXISTS "${QUPATHXX_VTI_LZ4_ARCHIVE}")
  message(FATAL_ERROR "Supply the admitted LZ4 1.10.0 library source ZIP")
endif()
file(SHA256 "${QUPATHXX_VTI_LZ4_ARCHIVE}" lz4_digest)
if(NOT lz4_digest STREQUAL "e9fb04130b7fc33727fa9e8e747ed067a8a0e7bef4b3769d689e2953a245d9c0")
  message(FATAL_ERROR "Unexpected LZ4 source archive")
endif()
set(lz4 "${CMAKE_CURRENT_BINARY_DIR}/lz4-inputs/lz4-1.10.0/lib")
if(NOT EXISTS "${lz4}/lz4.c")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_VTI_LZ4_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/lz4-inputs")
endif()
target_sources(qpx_vti_plugin PRIVATE "${lz4}/lz4.c")
target_include_directories(qpx_vti_plugin PRIVATE "${lz4}")
install(FILES "${lz4}/LICENSE" DESTINATION licenses/lz4)

set(QUPATHXX_VTI_XZ_ARCHIVE "" CACHE FILEPATH "Admitted offline XZ Embedded decoder source ZIP")
if(NOT EXISTS "${QUPATHXX_VTI_XZ_ARCHIVE}")
  message(FATAL_ERROR "Supply the admitted XZ Embedded decoder source ZIP")
endif()
file(SHA256 "${QUPATHXX_VTI_XZ_ARCHIVE}" xz_digest)
if(NOT xz_digest STREQUAL "4b83ab578420027fb7ba8c247d5f98d844df150bccb8f18bca48b7de2bed0c19")
  message(FATAL_ERROR "Unexpected XZ Embedded source archive")
endif()
set(xz "${CMAKE_CURRENT_BINARY_DIR}/xz-inputs/${xz_digest}")
set(xz_sources "${xz}/linux/lib/xz")
if(NOT EXISTS "${xz_sources}/xz_dec_stream.c")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_VTI_XZ_ARCHIVE}" DESTINATION "${xz}")
endif()
set(xz_units "${xz_sources}/xz_crc32.c" "${xz_sources}/xz_dec_lzma2.c" "${xz_sources}/xz_dec_stream.c")
set_source_files_properties(${xz_units} PROPERTIES COMPILE_DEFINITIONS XZ_DEC_SINGLE)
target_sources(qpx_vti_plugin PRIVATE ${xz_units})
target_include_directories(qpx_vti_plugin PRIVATE "${xz}/linux/include/linux" "${xz}/userspace")
install(FILES "${xz}/COPYING" "${xz}/AUTHORS" DESTINATION licenses/xz-embedded)
