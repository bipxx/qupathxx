cmake_minimum_required(VERSION 3.24)
project(qupathxx_png_reader_plugin VERSION 1.0.0 LANGUAGES C CXX)

set(QUPATHXX_READER_PLUGIN_KIT_PREFIX "" CACHE PATH "Relocated ReaderPluginKit prefix")
set(QUPATHXX_PNG_SOURCE_DIR "" CACHE PATH "Verified libpng core source selection")
set(QUPATHXX_PNG_ZLIB_SOURCE_DIR "" CACHE PATH "Admitted zlib 1.3.2 source directory")
option(QUPATHXX_PNG_BUILD_TESTS "Build independent ABI and installed-SDK checks" OFF)
foreach(name QUPATHXX_READER_PLUGIN_KIT_PREFIX QUPATHXX_PNG_SOURCE_DIR QUPATHXX_PNG_ZLIB_SOURCE_DIR)
  if(NOT IS_DIRECTORY "${${name}}")
    message(FATAL_ERROR "Set ${name} to its existing admitted input directory; this build never downloads dependencies")
  endif()
endforeach()
set(qpx_kit "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/share/qupathxx/reader-plugin")
include("${qpx_kit}/profile.cmake")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
   NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "This plugin requires Windows x64 MSVC")
endif()
if(NOT MSVC_VERSION STREQUAL QUPATHXX_READER_PLUGIN_MSVC_VERSION OR
   NOT CMAKE_CXX_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION OR
   NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the C/C++ compiler recorded by the ReaderPluginKit")
endif()
if(CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Supported configuration" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "This plugin requires CMAKE_BUILD_TYPE=Release")
endif()
if(CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreadedDLL")
  message(FATAL_ERROR "This plugin requires the Release dynamic CRT (/MD)")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/libpng-source-lock.json" png_lock)
string(JSON png_commit GET "${png_lock}" commit)
if(NOT png_commit STREQUAL "cd952f49f95bb27154ae77dbb103032d95f6e580")
  message(FATAL_ERROR "Unexpected libpng source identity")
endif()
string(JSON png_count LENGTH "${png_lock}" members)
math(EXPR png_last "${png_count}-1")
foreach(index RANGE 0 ${png_last})
  string(JSON member MEMBER "${png_lock}" members ${index})
  string(JSON expected GET "${png_lock}" members "${member}" sha256)
  if(NOT EXISTS "${QUPATHXX_PNG_SOURCE_DIR}/${member}")
    message(FATAL_ERROR "Missing libpng source: ${member}")
  endif()
  file(SHA256 "${QUPATHXX_PNG_SOURCE_DIR}/${member}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Changed libpng source: ${member}")
  endif()
endforeach()
file(READ "${QUPATHXX_PNG_ZLIB_SOURCE_DIR}/zlib.h" zlib_header)
if(NOT zlib_header MATCHES "#define ZLIB_VERSION \"1\\.3\\.2\"")
  message(FATAL_ERROR "The admitted compression source must be zlib 1.3.2")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/zlib-source-lock.json" zlib_lock)
string(JSON zlib_count LENGTH "${zlib_lock}" members)
math(EXPR zlib_last "${zlib_count}-1")
foreach(index RANGE 0 ${zlib_last})
  string(JSON member MEMBER "${zlib_lock}" members ${index})
  string(JSON expected GET "${zlib_lock}" members "${member}" sha256)
  if(NOT EXISTS "${QUPATHXX_PNG_ZLIB_SOURCE_DIR}/${member}")
    message(FATAL_ERROR "Missing zlib source: ${member}")
  endif()
  file(SHA256 "${QUPATHXX_PNG_ZLIB_SOURCE_DIR}/${member}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Changed zlib source: ${member}")
  endif()
endforeach()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
foreach(name GVMAT64 INFBACK9 CRC32VX)
  set(ZLIB_WITH_${name} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(name ADA BLAST IOSTREAM3 MINIZIP PUFF TESTZLIB ZLIB1_DLL)
  set(ZLIB_BUILD_${name} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${QUPATHXX_PNG_ZLIB_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/zlib")
file(READ "${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" zlib_configured)
if(NOT zlib_configured MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "PNG's private compression library must retain prefixed symbols")
endif()
# Consumers of zlib.h otherwise find its adjacent source zconf.h first, which
# has no configured prefix. Keep both public headers together, as the host's
# pinned compression recipe does, without editing upstream source files.
set(png_zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/png-include")
file(MAKE_DIRECTORY "${png_zlib_headers}")
configure_file("${QUPATHXX_PNG_ZLIB_SOURCE_DIR}/zlib.h" "${png_zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${png_zlib_headers}/zconf.h" COPYONLY)

configure_file("${QUPATHXX_PNG_SOURCE_DIR}/scripts/pnglibconf.h.prebuilt"
  "${CMAKE_CURRENT_BINARY_DIR}/pnglibconf.h" COPYONLY)
set(png_sources png.c pngerror.c pngget.c pngmem.c pngpread.c pngread.c pngrio.c
  pngrtran.c pngrutil.c pngset.c pngtrans.c pngwio.c pngwrite.c pngwtran.c pngwutil.c)
list(TRANSFORM png_sources PREPEND "${QUPATHXX_PNG_SOURCE_DIR}/")
add_library(qpx_png_codec STATIC ${png_sources})
target_include_directories(qpx_png_codec PUBLIC "${QUPATHXX_PNG_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" "${png_zlib_headers}")
target_compile_definitions(qpx_png_codec PUBLIC PNG_STATIC PRIVATE
  PNG_ARM_NEON_OPT=0 PNG_RISCV_RVV_OPT=0 PNG_MIPS_MSA_OPT=0 PNG_MIPS_MMI_OPT=0
  PNG_POWERPC_VSX_OPT=0 PNG_LOONGARCH_LSX_OPT=0 PNG_INTEL_SSE_OPT=0
  _CRT_SECURE_NO_WARNINGS)
target_link_libraries(qpx_png_codec PUBLIC zlibstatic)

configure_file(reader_plugin_profile.hpp.in reader_plugin_profile.hpp @ONLY)
add_library(qpx_png_plugin MODULE png_reader.cpp png_decoder.c)
set_target_properties(qpx_png_plugin PROPERTIES PREFIX "")
target_compile_definitions(qpx_png_plugin PRIVATE QPX_READER_PLUGIN_EXPORTS)
target_compile_options(qpx_png_plugin PRIVATE /utf-8 /W4 $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
target_include_directories(qpx_png_plugin PRIVATE
  "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(qpx_png_plugin PRIVATE qpx_png_codec)
add_custom_command(TARGET qpx_png_plugin POST_BUILD COMMAND "${CMAKE_COMMAND}"
  "-DMODULE_FILE=$<TARGET_FILE:qpx_png_plugin>"
  "-DPROFILE=windows-x64-msvc${QUPATHXX_READER_PLUGIN_MSVC_VERSION}-release-md"
  "-DOUTPUT=$<TARGET_FILE_DIR:qpx_png_plugin>/reader-plugins.json"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/write_manifest.cmake" VERBATIM)

install(TARGETS qpx_png_plugin RUNTIME DESTINATION . LIBRARY DESTINATION . CONFIGURATIONS Release)
install(FILES "$<TARGET_FILE_DIR:qpx_png_plugin>/reader-plugins.json" DESTINATION . CONFIGURATIONS Release)
install(FILES "${QUPATHXX_PNG_SOURCE_DIR}/LICENSE" DESTINATION licenses/libpng)
install(FILES "${QUPATHXX_PNG_ZLIB_SOURCE_DIR}/LICENSE" DESTINATION licenses/zlib)
install(FILES "${qpx_kit}/LICENSE" DESTINATION licenses/qupathxx)

if(QUPATHXX_PNG_BUILD_TESTS)
  add_executable(png_plugin_tests tests/png_plugin_tests.cpp)
  target_include_directories(png_plugin_tests PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include")
  target_compile_options(png_plugin_tests PRIVATE /utf-8 /permissive-)
  find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
  add_executable(png_project_consumer tests/png_project_consumer.cpp)
  target_compile_options(png_project_consumer PRIVATE /utf-8 /permissive-)
  target_link_libraries(png_project_consumer PRIVATE qupathxx::core)
  qupathxx_stage_runtime(png_project_consumer)
endif()
