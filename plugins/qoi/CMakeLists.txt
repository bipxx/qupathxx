cmake_minimum_required(VERSION 3.24)
project(qupathxx_qoi_reader_plugin VERSION 1.0.0 LANGUAGES CXX)
set(qpx_image_target qpx_qoi_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/qoi_reader.cpp")
set(qpx_image_id org.qupathxx.qoi)
set(qpx_image_name QOI)
set(qpx_image_contract qoi-rgba8-native-samples/1)
include("../image-common/plugin_build.cmake")

option(QUPATHXX_QOI_BUILD_TESTS "Build independent C ABI and native SDK consumers" OFF)
if(QUPATHXX_QOI_BUILD_TESTS)
  add_executable(qoi_plugin_tests tests/qoi_plugin_tests.cpp)
  target_include_directories(qoi_plugin_tests PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include")
  target_compile_options(qoi_plugin_tests PRIVATE /utf-8 /W4 /permissive-)
  find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
  add_executable(qoi_project_consumer tests/qoi_project_consumer.cpp)
  target_compile_options(qoi_project_consumer PRIVATE /utf-8 /W4 /permissive-)
  target_link_libraries(qoi_project_consumer PRIVATE qupathxx::core)
  qupathxx_stage_runtime(qoi_project_consumer)
  set(QUPATHXX_QOI_ORACLE_INCLUDE "" CACHE PATH "Optional pinned development-only MIT qoi.h directory")
  if(QUPATHXX_QOI_ORACLE_INCLUDE)
    if(NOT EXISTS "${QUPATHXX_QOI_ORACLE_INCLUDE}/qoi.h")
      message(FATAL_ERROR "Supply the exact pinned development oracle locally; no download fallback")
    endif()
    add_executable(qoi_reference tests/qoi_reference.cpp)
    target_include_directories(qoi_reference PRIVATE "${QUPATHXX_QOI_ORACLE_INCLUDE}")
    target_compile_options(qoi_reference PRIVATE /utf-8 /W4 /permissive-)
  endif()
endif()
