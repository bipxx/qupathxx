cmake_minimum_required(VERSION 3.24)
project(qupathxx_fits_reader_plugin VERSION 1.9.0 LANGUAGES C CXX)

set(qpx_image_target qpx_fits_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/fits_reader.cpp")
set(qpx_image_id org.qupathxx.fits)
set(qpx_image_name FITS)
set(qpx_image_contract fits-image-hdu-scalar-window/2)
include("../image-common/plugin_build.cmake")
target_compile_definitions(qpx_fits_plugin PRIVATE QPX_FITS_VERSION="${PROJECT_VERSION}")

set(QUPATHXX_FITS_ZLIB_ARCHIVE "" CACHE FILEPATH "Admitted offline zlib 1.3.2 source archive")
if(NOT EXISTS "${QUPATHXX_FITS_ZLIB_ARCHIVE}")
  message(FATAL_ERROR "Provide the admitted offline zlib 1.3.2 source archive.")
endif()
file(SHA256 "${QUPATHXX_FITS_ZLIB_ARCHIVE}" zlib_digest)
if(NOT zlib_digest STREQUAL "bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16")
  message(FATAL_ERROR "FITS requires the exact admitted zlib archive")
endif()
if(NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the C compiler recorded by the ReaderPluginKit")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(zlib_root "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs")
set(zlib_source "${zlib_root}/zlib-1.3.2")
if(NOT EXISTS "${zlib_source}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_FITS_ZLIB_ARCHIVE}" DESTINATION "${zlib_root}")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
add_subdirectory("${zlib_source}" "${CMAKE_CURRENT_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
set(zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/include")
configure_file("${zlib_source}/zlib.h" "${zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${zlib_headers}/zconf.h" COPYONLY)
file(READ "${zlib_headers}/zconf.h" zlib_config)
if(NOT zlib_config MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "FITS requires private prefixed zlib symbols")
endif()
target_include_directories(qpx_fits_plugin PRIVATE "${zlib_headers}")
target_link_libraries(qpx_fits_plugin PRIVATE zlibstatic)
install(FILES "${zlib_source}/LICENSE" DESTINATION licenses/zlib)

option(QUPATHXX_FITS_BUILD_TESTS "Build independent C ABI and native SDK consumers" OFF)
if(QUPATHXX_FITS_BUILD_TESTS)
  add_executable(fits_plugin_tests tests/fits_plugin_tests.cpp)
  target_include_directories(fits_plugin_tests PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include")
  target_compile_options(fits_plugin_tests PRIVATE /utf-8 /W4 /permissive-)

  find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
  add_executable(fits_project_consumer tests/fits_project_consumer.cpp)
  target_compile_options(fits_project_consumer PRIVATE /utf-8 /W4 /permissive-)
  target_link_libraries(fits_project_consumer PRIVATE qupathxx::core)
  qupathxx_stage_runtime(fits_project_consumer)
endif()
