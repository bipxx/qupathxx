cmake_minimum_required(VERSION 3.24)
project(qupathxx_nifti_reader_plugin VERSION 1.5.0 LANGUAGES C CXX)

set(qpx_image_target qpx_nifti_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/nifti_reader.cpp")
set(qpx_image_id org.qupathxx.nifti)
set(qpx_image_name NIfTI)
set(qpx_image_contract nifti-scalar-window/2)
set(qpx_image_source_files ON)
include("../image-common/plugin_build.cmake")
target_compile_definitions(qpx_nifti_plugin PRIVATE QPX_READER_PLUGIN_SOURCE_FILES
  QPX_NIFTI_VERSION="${PROJECT_VERSION}" QPX_NIFTI_CONTRACT="${qpx_image_contract}"
  QPX_NIFTI_NAME="${qpx_image_name}")

if(NOT CMAKE_C_COMPILER_VERSION STREQUAL QUPATHXX_READER_PLUGIN_CXX_COMPILER_VERSION)
  message(FATAL_ERROR "Use the C compiler recorded by the ReaderPluginKit")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(QUPATHXX_NIFTI_ZLIB_ARCHIVE "" CACHE FILEPATH "Admitted offline zlib 1.3.2 source archive")
if(NOT EXISTS "${QUPATHXX_NIFTI_ZLIB_ARCHIVE}")
  message(FATAL_ERROR "Provide the admitted zlib 1.3.2 source archive. No dependency downloads are enabled.")
endif()
file(SHA256 "${QUPATHXX_NIFTI_ZLIB_ARCHIVE}" zlib_digest)
if(NOT zlib_digest STREQUAL "bb329a0a2cd0274d05519d61c667c062e06990d72e125ee2dfa8de64f0119d16")
  message(FATAL_ERROR "NIfTI requires the exact admitted zlib archive")
endif()
set(zlib_root "${CMAKE_CURRENT_BINARY_DIR}/zlib-inputs")
set(zlib_sources "${zlib_root}/zlib-1.3.2")
if(NOT EXISTS "${zlib_sources}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_NIFTI_ZLIB_ARCHIVE}" DESTINATION "${zlib_root}")
endif()
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
set(ZLIB_PREFIX ON CACHE BOOL "" FORCE)
set(Z_PREFIX ON)
add_subdirectory("${zlib_sources}" "${CMAKE_CURRENT_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
set(zlib_headers "${CMAKE_CURRENT_BINARY_DIR}/zlib/include")
configure_file("${zlib_sources}/zlib.h" "${zlib_headers}/zlib.h" COPYONLY)
configure_file("${CMAKE_CURRENT_BINARY_DIR}/zlib/zconf.h" "${zlib_headers}/zconf.h" COPYONLY)
file(READ "${zlib_headers}/zconf.h" zlib_config)
if(NOT zlib_config MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "NIfTI requires private prefixed zlib symbols")
endif()
target_include_directories(qpx_nifti_plugin PRIVATE "${zlib_headers}")
target_link_libraries(qpx_nifti_plugin PRIVATE zlibstatic)
install(FILES "${zlib_sources}/LICENSE" DESTINATION licenses/zlib)

option(QUPATHXX_NIFTI_BUILD_TESTS "Build independent C ABI and native SDK consumers" OFF)
if(QUPATHXX_NIFTI_BUILD_TESTS)
  add_executable(nifti_plugin_tests tests/nifti_plugin_tests.cpp)
  target_include_directories(nifti_plugin_tests PRIVATE "${QUPATHXX_READER_PLUGIN_KIT_PREFIX}/include")
  target_compile_options(nifti_plugin_tests PRIVATE /utf-8 /W4 /permissive-)

  find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
  add_executable(nifti_project_consumer tests/nifti_project_consumer.cpp)
  target_compile_options(nifti_project_consumer PRIVATE /utf-8 /W4 /permissive-)
  target_link_libraries(nifti_project_consumer PRIVATE qupathxx::core)
  qupathxx_stage_runtime(nifti_project_consumer)
endif()
