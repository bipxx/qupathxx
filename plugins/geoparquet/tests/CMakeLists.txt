cmake_minimum_required(VERSION 3.24)
project(geoparquet_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(qupathxx CONFIG REQUIRED COMPONENTS core)
add_executable(geoparquet_probe project_probe.cpp)
target_compile_options(geoparquet_probe PRIVATE /utf-8)
target_link_libraries(geoparquet_probe PRIVATE qupathxx::core)
qupathxx_stage_runtime(geoparquet_probe)
