cmake_minimum_required(VERSION 3.24)
project(qupathxx_geopackage_plugin VERSION 1.1.0 LANGUAGES C CXX)
set(qpx_provider_version "${PROJECT_VERSION}")
set(qpx_provider_target qpx_geopackage_plugin)
set(qpx_provider_source geopackage.cpp)
set(qpx_provider_id org.qupathxx.geopackage)
set(qpx_format_id gpkg)
set(qpx_provider_domain annotation)
set(qpx_provider_abi 3)
set(qpx_provider_name "GeoPackage pixel annotations")
set(qpx_provider_contract gpkg-cartesian-image-pixel/2)
include("../format-common/plugin_build.cmake")

set(QUPATHXX_GPKG_SQLITE_ARCHIVE "" CACHE FILEPATH "Exact admitted SQLite 3.54.1 amalgamation ZIP")
file(SHA256 "${QUPATHXX_GPKG_SQLITE_ARCHIVE}" digest)
include("sqlite_pin.cmake")
if(NOT digest STREQUAL qpx_sqlite_archive_sha256)
  message(FATAL_ERROR "Use the exact admitted SQLite amalgamation")
endif()
set(sqlite "${CMAKE_CURRENT_BINARY_DIR}/sqlite/sqlite-amalgamation-3540100")
if(NOT EXISTS "${sqlite}/sqlite3.c")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_GPKG_SQLITE_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/sqlite")
endif()
foreach(member sqlite3.c sqlite3.h)
  file(SHA256 "${sqlite}/${member}" extracted)
  if(NOT extracted STREQUAL qpx_${member}_sha256)
    message(FATAL_ERROR "Extracted SQLite member differs from its pin")
  endif()
endforeach()
add_library(qpx_gpkg_sqlite STATIC "${sqlite}/sqlite3.c")
set_target_properties(qpx_gpkg_sqlite PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_definitions(qpx_gpkg_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION
  SQLITE_OMIT_SHARED_CACHE SQLITE_DQS=0)
target_include_directories(qpx_geopackage_plugin PRIVATE "${sqlite}")
target_link_libraries(qpx_geopackage_plugin PRIVATE qpx_gpkg_sqlite)
install(FILES "../../third_party/sqlite-LICENSE.txt" DESTINATION licenses/sqlite)
