function(unpack_archive stem digest destination)
  set(archive "${QUPATHXX_JPEGXL_ARCHIVES}/${stem}.tar.gz")
  if(NOT EXISTS "${archive}")
    message(FATAL_ERROR "Missing admitted archive ${stem}.tar.gz")
  endif()
  file(SHA256 "${archive}" actual)
  if(NOT actual STREQUAL digest)
    message(FATAL_ERROR "Unexpected source identity for ${stem}")
  endif()
  if(NOT EXISTS "${input}/${stem}.verified")
    file(MAKE_DIRECTORY "${destination}")
    execute_process(COMMAND "${ARCHIVE_TAR}" -xf "${archive}" -C "${destination}" ${ARGN}
      COMMAND_ERROR_IS_FATAL ANY)
    file(WRITE "${input}/${stem}.verified" "${actual}")
  endif()
endfunction()
