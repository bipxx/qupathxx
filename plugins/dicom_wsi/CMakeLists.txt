cmake_minimum_required(VERSION 3.24)
project(qupathxx_dicom_wsi_reader_plugin VERSION 1.0.0 LANGUAGES C CXX)

set(QUPATHXX_DICOM_WSI_SOURCE_ARCHIVE "" CACHE FILEPATH "Admitted OpenSlide 4.0.1.2 corresponding-source archive")
set(QUPATHXX_DICOM_WSI_OPENSLIDE_SDK "" CACHE PATH "Exact OpenSlide 4.0.1.2 Windows x64 SDK")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" lock)

function(check_locked_file file expected)
  file(SHA256 "${file}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "DICOM WSI build input differs from its source lock: ${file}")
  endif()
endfunction()

string(JSON archive_digest GET "${lock}" carrier sha256)
string(JSON prefix GET "${lock}" carrier prefix)
check_locked_file("${QUPATHXX_DICOM_WSI_SOURCE_ARCHIVE}" "${archive_digest}")
set(extract_dir "${CMAKE_CURRENT_BINARY_DIR}/dicom-inputs")
set(input_root "${extract_dir}/${prefix}")
set(patterns "")
set(metadata_paths "")
string(JSON count LENGTH "${lock}" metadata_source_members)
math(EXPR last "${count}-1")
foreach(i RANGE 0 "${last}")
  string(JSON member MEMBER "${lock}" metadata_source_members "${i}")
  list(APPEND patterns "${prefix}${member}")
  list(APPEND metadata_paths "${member}")
endforeach()
if(NOT EXISTS "${input_root}/libdicom-1.3.0/LICENSE")
  file(ARCHIVE_EXTRACT INPUT "${QUPATHXX_DICOM_WSI_SOURCE_ARCHIVE}"
       DESTINATION "${extract_dir}" PATTERNS ${patterns})
endif()
foreach(member IN LISTS metadata_paths)
  string(JSON expected GET "${lock}" metadata_source_members "${member}" sha256)
  check_locked_file("${input_root}/${member}" "${expected}")
endforeach()

set(sdk "${QUPATHXX_DICOM_WSI_OPENSLIDE_SDK}")
string(JSON count LENGTH "${lock}" openslide_sdk members)
math(EXPR last "${count}-1")
foreach(i RANGE 0 "${last}")
  string(JSON member MEMBER "${lock}" openslide_sdk members "${i}")
  string(JSON expected GET "${lock}" openslide_sdk members "${member}")
  check_locked_file("${sdk}/${member}" "${expected}")
  if(member MATCHES "^licenses/(.*)/[^/]+$")
    install(FILES "${sdk}/${member}" DESTINATION "licenses/openslide-sdk/${CMAKE_MATCH_1}")
  endif()
endforeach()

set(qpx_image_target qpx_dicom_wsi_plugin)
set(qpx_image_source "${CMAKE_CURRENT_SOURCE_DIR}/dicom_wsi_reader.cpp")
set(qpx_image_id org.qupathxx.dicom-wsi)
set(qpx_image_name "DICOM WSI")
set(qpx_image_contract dicom-wsi-openslide/1)
set(qpx_image_manifest_name dicom-wsi-reader-plugins.json)
set(qpx_image_source_files ON)
set(qpx_image_display_only ON)
set(qpx_image_sdk_runtime_files "${sdk}/bin/libopenslide-1.dll")
include("../image-common/plugin_build.cmake")

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(dcm "${input_root}/libdicom-1.3.0")
set(ut "${input_root}/uthash-2.3.0/src")
set(gen "${CMAKE_CURRENT_BINARY_DIR}/dicom-generated")
file(MAKE_DIRECTORY "${gen}")
file(CONFIGURE OUTPUT "${gen}/config.h" CONTENT "#define HAVE_IO_H 1\n" @ONLY)
set(DCM_VERSION 1.3.0)
set(DCM_SUFFIXED_VERSION 1.3.0)
set(DCM_VERSION_MAJOR 1)
set(DCM_VERSION_MINOR 3)
set(DCM_VERSION_MICRO 0)
set(DCM_ABI_VERSION_MAJOR 1)
set(DCM_ABI_VERSION_MINOR 3)
set(DCM_ABI_VERSION_PATCH 0)
set(DCM_STATIC 1)
configure_file("${dcm}/include/dicom/version.h.in" "${gen}/version.h" @ONLY)
add_executable(qpx_dicom_dictionary "${dcm}/src/dicom-dict-build.c" "${dcm}/src/dicom-dict-tables.c")
target_include_directories(qpx_dicom_dictionary PRIVATE "${dcm}/include" "${gen}" "${ut}")
add_custom_command(OUTPUT "${gen}/dicom-dict-lookup.c" "${gen}/dicom-dict-lookup.h"
  COMMAND qpx_dicom_dictionary "${gen}/dicom-dict-lookup.c" "${gen}/dicom-dict-lookup.h" VERBATIM)
add_library(qpx_dicom_metadata STATIC "${gen}/dicom-dict-lookup.c")
foreach(unit dicom-data dicom-dict-tables dicom-dict dicom-file dicom-io dicom-parse dicom)
  target_sources(qpx_dicom_metadata PRIVATE "${dcm}/src/${unit}.c")
endforeach()
target_include_directories(qpx_dicom_metadata PUBLIC "${dcm}/include" "${gen}" PRIVATE "${dcm}/src" "${ut}")
target_compile_definitions(qpx_dicom_metadata PRIVATE BUILDING_LIBDICOM)
target_compile_definitions(qpx_dicom_wsi_plugin PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN QPX_READER_PLUGIN_SOURCE_FILES)
target_include_directories(qpx_dicom_wsi_plugin PRIVATE "${sdk}/include/openslide")
target_link_libraries(qpx_dicom_wsi_plugin PRIVATE qpx_dicom_metadata "${sdk}/lib/libopenslide.lib")

install(FILES "${dcm}/LICENSE" DESTINATION licenses/libdicom)
install(FILES "${input_root}/uthash-2.3.0/LICENSE" DESTINATION licenses/uthash)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" DESTINATION .)
