cmake_minimum_required(VERSION 3.24)
project(qupathxx_anndata_zarr_plugin VERSION 1.2.0 LANGUAGES CXX)

set(qpx_provider_target qpx_anndata_zarr_plugin)
set(qpx_provider_source anndata_zarr.cpp)
set(qpx_provider_id org.qupathxx.anndata-zarr)
set(qpx_format_id anndata-zarr)
set(qpx_provider_domain matrix)
set(qpx_provider_name "AnnData Zarr typed expression matrix")
set(qpx_provider_contract anndata-zarr2-typed/1)
set(qpx_provider_version "${PROJECT_VERSION}")
set(qpx_provider_abi 4)
set(qpx_provider_manifest_name anndata-zarr-format-plugins.json)
include("${CMAKE_CURRENT_SOURCE_DIR}/../format-common/plugin_build.cmake")

# Read the exact retained SDK only to locate its admitted prefixed zlib archive.
# No application core object, Python, Java or arbitrary installed codec is linked.
find_package(qupathxx 0.1.0 CONFIG REQUIRED COMPONENTS core)
get_target_property(qpx_anndata_core qupathxx::core IMPORTED_LOCATION_RELEASE)
if(NOT EXISTS "${qpx_anndata_core}")
  message(FATAL_ERROR "AnnData Zarr requires the exact retained FP04 SDK context")
endif()
file(SHA256 "${qpx_anndata_core}" qpx_anndata_core_sha)
if(NOT qpx_anndata_core_sha STREQUAL "4bf957c9ea6e46294c11b9bfb7d9c2496cf4fca607a9c28ba837768d301eb6bb")
  message(FATAL_ERROR "Use the recorded frozen SDK context; qualify other products separately")
endif()
get_filename_component(qpx_anndata_libdir "${qpx_anndata_core}" DIRECTORY)
get_filename_component(qpx_anndata_sdk "${qpx_anndata_libdir}/../.." ABSOLUTE)
set(qpx_anndata_zlib "${qpx_anndata_libdir}/deps/qpx-zlib.lib")
if(NOT EXISTS "${qpx_anndata_zlib}")
  message(FATAL_ERROR "Missing frozen prefixed zlib1.3.2 static library")
endif()
file(SHA256 "${qpx_anndata_zlib}" qpx_anndata_zlib_sha)
if(NOT qpx_anndata_zlib_sha STREQUAL "4ec525ee88a54ccf5d33ef1a5c8105533ad19f0ac946b71b52c8a824cbefb63d")
  message(FATAL_ERROR "The selected zlib static product differs")
endif()
set(QUPATHXX_ANNDATA_ZARR_ZLIB_INCLUDE "" CACHE PATH "Matching frozen prefixed zlib1.3.2 paired qpx-include")
if(NOT EXISTS "${QUPATHXX_ANNDATA_ZARR_ZLIB_INCLUDE}/zlib.h" OR
   NOT EXISTS "${QUPATHXX_ANNDATA_ZARR_ZLIB_INCLUDE}/zconf.h")
  message(FATAL_ERROR "Provide the exact frozen paired zlib public headers")
endif()
file(READ "${QUPATHXX_ANNDATA_ZARR_ZLIB_INCLUDE}/zconf.h" qpx_anndata_zconf)
if(NOT qpx_anndata_zconf MATCHES "#define Z_PREFIX 1")
  message(FATAL_ERROR "The selected static zlib requires matching prefixed headers")
endif()
target_include_directories(qpx_anndata_zarr_plugin PRIVATE "${QUPATHXX_ANNDATA_ZARR_ZLIB_INCLUDE}")
target_compile_definitions(qpx_anndata_zarr_plugin PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(qpx_anndata_zarr_plugin PRIVATE "${qpx_anndata_zlib}")

# Reuse the exact admitted C-Blosc/LZ4 static product from the optional byte
# provider. AnnData's store reader must decode already-observed bytes in memory.
set(QUPATHXX_ANNDATA_ZARR_BLOSC_LIBRARY "" CACHE FILEPATH "Verified qpx_blosc.lib from the Blosc byte provider")
set(QUPATHXX_ANNDATA_ZARR_BLOSC_INCLUDE "" CACHE PATH "Matching C-Blosc 1.21.6 header directory")
set(qpx_blosc_inputs "${CMAKE_CURRENT_SOURCE_DIR}/../blosc-bytes")
file(READ "${qpx_blosc_inputs}/source-lock.json" qpx_blosc_lock)
foreach(header IN ITEMS blosc.h blosc-export.h)
  string(JSON expected GET "${qpx_blosc_lock}" members "blosc/${header}")
  file(SHA256 "${QUPATHXX_ANNDATA_ZARR_BLOSC_INCLUDE}/${header}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "C-Blosc header does not match the admitted source: ${header}")
  endif()
endforeach()
file(SHA256 "${QUPATHXX_ANNDATA_ZARR_BLOSC_LIBRARY}" qpx_blosc_sha)
if(NOT qpx_blosc_sha STREQUAL "804c02c3b167de146c3c18b7caa3511b237f23fafbbef8fb12936736618c4cf6")
  message(FATAL_ERROR "The selected C-Blosc static product differs from the admitted build")
endif()
target_include_directories(qpx_anndata_zarr_plugin PRIVATE "${QUPATHXX_ANNDATA_ZARR_BLOSC_INCLUDE}")
target_link_libraries(qpx_anndata_zarr_plugin PRIVATE "${QUPATHXX_ANNDATA_ZARR_BLOSC_LIBRARY}")
install(FILES "${qpx_blosc_inputs}/source-lock.json" "${qpx_blosc_inputs}/THIRD_PARTY_NOTICES.txt"
  DESTINATION licenses/blosc)
install(FILES "${qpx_anndata_sdk}/share/qupathxx/licenses/tiff-zlib/LICENSE" DESTINATION licenses/zlib)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.txt"
              "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" DESTINATION .)
